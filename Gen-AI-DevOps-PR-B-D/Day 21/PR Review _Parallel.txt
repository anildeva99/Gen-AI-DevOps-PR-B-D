#!/usr/bin/env pwsh
# ==========================================
# 🤖 Azure DevOps + Azure OpenAI PR Review (Parallel + Safe Fallback + Timestamps)
# ==========================================

# 🕓 Timestamp logger function
function Log($msg) {
    Write-Host ("[{0}] {1}" -f (Get-Date -Format "HH:mm:ss.fff"), $msg)
}

# 1️⃣ Validate environment variables
$requiredVars = @(
    "AZDO_ORG_URL", "AZDO_PROJECT", "AZDO_REPO_ID", "AZDO_PAT",
    "OPENAI_API_KEY", "OPENAI_ENDPOINT", "OPENAI_DEPLOYMENT_NAME"
)
foreach ($var in $requiredVars) {
    if (-not (Get-Item -Path "Env:$var" -ErrorAction SilentlyContinue)) {
        Log "❌ Missing required environment variable: $var"
        exit 1
    }
}

# 2️⃣ Setup headers
$devOpsHeaders = @{
    Authorization = "Basic " + [Convert]::ToBase64String(
        [Text.Encoding]::ASCII.GetBytes(":$($env:AZDO_PAT)")
    )
    "Content-Type" = "application/json"
}
$openAIHeaders = @{
    "api-key" = $env:OPENAI_API_KEY
    "Content-Type" = "application/json"
}

# 3️⃣ Fetch all active PRs
Log "🔹 Fetching all active PRs..."
$prUri = "$($env:AZDO_ORG_URL)/$($env:AZDO_PROJECT)/_apis/git/repositories/$($env:AZDO_REPO_ID)/pullrequests?searchCriteria.status=active&api-version=7.0"
$prResponse = Invoke-RestMethod -Uri $prUri -Headers $devOpsHeaders

if (-not $prResponse.value) {
    Log "⚠️ No active PRs found. Exiting."
    exit
}

$allPRs = $prResponse.value | Sort-Object creationDate -Descending
Log "🔹 Total Active PRs Found: $($allPRs.Count)"

# 4️⃣ Define PR review logic (parallel-safe)
$reviewBlock = {
    param($pr, $devOpsHeaders, $openAIHeaders, $envVars)

    function Log($msg) {
        Write-Host ("[{0}] {1}" -f (Get-Date -Format "HH:mm:ss.fff"), $msg)
    }

    $orgUrl  = $envVars.ORG
    $project = $envVars.PROJECT
    $repoId  = $envVars.REPO
    $endpoint = $envVars.OPENAI_ENDPOINT
    $deploy = $envVars.OPENAI_DEPLOYMENT

    try {
        $prId = $pr.pullRequestId
        $commitId = $pr.lastMergeSourceCommit.commitId
        Log "🔸 Processing PR ID: $prId"
        Log "   Latest commit: $commitId"

        # Fetch changed files
        $changesUri = "$orgUrl/$project/_apis/git/repositories/$repoId/pullRequests/$prId/iterations/1/changes?api-version=7.0"
        $changesResponse = Invoke-RestMethod -Uri $changesUri -Headers $devOpsHeaders
        $pythonFiles = $changesResponse.changeEntries | Where-Object { $_.item.path -like "*.py" }

        # Save debug
        $changesResponse | ConvertTo-Json -Depth 5 | Out-File -Encoding UTF8 "changes_debug_pr_$prId.json"
        Log "💾 Saved changes_debug_pr_$prId.json"

        foreach ($file in $pythonFiles) {
            $filePath = $file.item.path
            Log "📄 [$prId] Found Python file: $filePath"

            # Fetch file content
            $fileUri = "$orgUrl/$project/_apis/git/repositories/$repoId/items?path=$filePath&version=$commitId&versionType=commit&includeContent=true&api-version=7.0&`$format=text"
            try {
                $content = Invoke-RestMethod -Uri $fileUri -Headers $devOpsHeaders
            } catch {
                Log "⚠️ [$prId] Could not fetch content for $filePath"
                continue
            }

            # 🧹 Syntax + Lint check
            $temp = [System.IO.Path]::GetTempFileName() + ".py"
            Set-Content $temp $content
            $syntaxIssue = $false
            try { python3 -m py_compile $temp } catch { $syntaxIssue = $true }
            $lintOutput = python3 -m pylint $temp --score=no 2>&1
            Remove-Item $temp -Force

            # 🤖 AI Review
            $body = @{
                messages = @(
                    @{ role = "system"; content = "You are a senior Python reviewer. Check for syntax, logic, and security issues." },
                    @{ role = "user"; content = "Review this Python file:`n$content" }
                )
            } | ConvertTo-Json -Depth 4

            try {
                $aiUri = "$endpoint/openai/deployments/$deploy/chat/completions?api-version=2024-02-01"
                $aiResponse = Invoke-RestMethod -Uri $aiUri -Headers $openAIHeaders -Method Post -Body $body
                $aiFeedback = $aiResponse.choices[0].message.content
                Log "🤖 [$prId] AI review received."
            } catch {
                $aiFeedback = "⚠️ [$prId] AI review failed: $_"
            }

            # 📝 Post PR Comment
            $comment = "🤖 **AI Review for ${filePath}:**`n$aiFeedback"
            if ($syntaxIssue -or $lintOutput) {
                $comment += "`n`n⚠️ **Lint/Syntax Issues:**`n$lintOutput"
            }

            $commentBody = @{
                comments = @(@{ content = $comment; commentType = "text" })
                status = "active"
            } | ConvertTo-Json -Depth 4

            $commentUri = "$orgUrl/$project/_apis/git/repositories/$repoId/pullRequests/$prId/threads?api-version=7.0"
            Invoke-RestMethod -Uri $commentUri -Headers $devOpsHeaders -Method Post -Body $commentBody

            Log "✅ [$prId] Comment added for $filePath"
        }

        Log "✔ [$prId] Review complete."
    }
    catch {
        Log "❌ [$($pr.pullRequestId)] Error: $_"
    }
}

# 5️⃣ Run in parallel with diagnostics
try {
    Log "⚡ Running PR reviews in parallel..."
    $throttle = 3

    $envVars = @{
        ORG = $env:AZDO_ORG_URL
        PROJECT = $env:AZDO_PROJECT
        REPO = $env:AZDO_REPO_ID
        OPENAI_ENDPOINT = $env:OPENAI_ENDPOINT
        OPENAI_DEPLOYMENT = $env:OPENAI_DEPLOYMENT_NAME
    }

    $jobs = foreach ($pr in $allPRs) {
        Start-Job -ScriptBlock $reviewBlock -ArgumentList $pr, $devOpsHeaders, $openAIHeaders, $envVars
    }

    Log "⚙️ Total Background Jobs Started: $($jobs.Count)"
    $jobs | Format-Table Id, Name, State, PSBeginTime

    Log "⏳ Waiting for all PR review jobs to complete..."
    $jobs | Wait-Job | Out-Null

    Log "📋 Collecting all job results..."
    $jobs | ForEach-Object { Receive-Job $_ | Write-Host }

    # 🧾 Job summary with timing
    Log "🧾 Job Summary:"
    $jobs | Select-Object Id, Name, State, PSBeginTime, PSEndTime | Format-Table

    # Cleanup
    $jobs | Remove-Job -Force
    Log "🎯 All PRs processed successfully in parallel mode."
}
catch {
    Log "⚠️ Parallel mode failed, switching to sequential fallback..."
    foreach ($pr in $allPRs) {
        & $reviewBlock $pr $devOpsHeaders $openAIHeaders $envVars
    }
    Log "🎯 All PRs processed sequentially (fallback mode)."
}
