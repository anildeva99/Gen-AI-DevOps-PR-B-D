#!/usr/bin/env pwsh
# ==========================================
# 🔹 Azure DevOps + Azure OpenAI PR Review (Queue Processing)
# ==========================================

# 1️⃣ Check required environment variables
$requiredVars = @(
    "AZDO_ORG_URL", "AZDO_PROJECT", "AZDO_REPO_ID", "AZDO_PAT",
    "OPENAI_API_KEY", "OPENAI_ENDPOINT", "OPENAI_DEPLOYMENT_NAME"
)
foreach ($var in $requiredVars) {
    if (-not (Get-Item -Path "Env:$var" -ErrorAction SilentlyContinue)) {
        Write-Host "❌ Missing required environment variable: $var"
        exit 1
    }
}

# 2️⃣ Set headers
$devOpsHeaders = @{
    Authorization = "Basic " + [Convert]::ToBase64String(
        [Text.Encoding]::ASCII.GetBytes(":$($env:AZDO_PAT)")
    )
    "Content-Type" = "application/json"
}
$openAIHeaders = @{
    "api-key" = $env:OPENAI_API_KEY
    "Content-Type" = "application/json"
}

# 3️⃣ Fetch all active PRs (in queue)
Write-Host "🔹 Fetching all active PRs..."
$prUri = "$($env:AZDO_ORG_URL)/$($env:AZDO_PROJECT)/_apis/git/repositories/$($env:AZDO_REPO_ID)/pullrequests?searchCriteria.status=active&api-version=7.0"
$prResponse = Invoke-RestMethod -Uri $prUri -Headers $devOpsHeaders

if (-not $prResponse.value) {
    Write-Host "⚠️ No active PRs found. Exiting."; exit
}

$allPRs = $prResponse.value | Sort-Object creationDate -Descending
Write-Host "🔹 Total Active PRs Found: $($allPRs.Count)"

# 4️⃣ Process PRs one by one
foreach ($pr in $allPRs) {
    $prId = $pr.pullRequestId
    $commitId = $pr.lastMergeSourceCommit.commitId

    Write-Host "`n🔸 Processing PR ID: $prId"
    Write-Host "   Latest commit: $commitId"

    # Get files changed in PR
    $changesUri = "$($env:AZDO_ORG_URL)/$($env:AZDO_PROJECT)/_apis/git/repositories/$($env:AZDO_REPO_ID)/pullRequests/$prId/iterations/1/changes?api-version=7.0"
    $changesResponse = Invoke-RestMethod -Uri $changesUri -Headers $devOpsHeaders
    $pythonFiles = $changesResponse.changeEntries | Where-Object { $_.item.path -like "*.py" }

    # Save debug
    $changesResponse | ConvertTo-Json -Depth 5 | Out-File -Encoding UTF8 "changes_debug_pr_$prId.json"
    Write-Host "   💾 Saved changes_debug_pr_$prId.json"

    foreach ($file in $pythonFiles) {
        $filePath = $file.item.path
        Write-Host "📄 Found Python file: $filePath"

        # Fetch file content
        $fileUri = "$($env:AZDO_ORG_URL)/$($env:AZDO_PROJECT)/_apis/git/repositories/$($env:AZDO_REPO_ID)/items?path=$filePath&version=$commitId&versionType=commit&includeContent=true&api-version=7.0&`$format=text"
        try {
            $content = Invoke-RestMethod -Uri $fileUri -Headers $devOpsHeaders
        } catch {
            Write-Host "⚠️ Could not fetch content for $filePath"; continue
        }

        # ⛏ Run Syntax + Linter
        $temp = [System.IO.Path]::GetTempFileName() + ".py"
        Set-Content $temp $content
        $syntaxIssue = $false
        try { python3 -m py_compile $temp } catch { $syntaxIssue = $true }
        $lintOutput = python3 -m pylint $temp --score=no 2>&1
        Remove-Item $temp -Force

        # 🤖 AI Review using Azure OpenAI
        $body = @{
            messages = @(
                @{ role = "system"; content = "You are a senior Python reviewer. Find errors, bad logic, unsafe code, and suggest improvements clearly." },
                @{ role = "user"; content = "Review this code:`n$content" }
            )
        } | ConvertTo-Json -Depth 4

        try {
            $aiUri = "$($env:OPENAI_ENDPOINT)openai/deployments/$($env:OPENAI_DEPLOYMENT_NAME)/chat/completions?api-version=2024-02-01"
            $aiResponse = Invoke-RestMethod -Uri $aiUri -Headers $openAIHeaders -Method Post -Body $body
            $aiFeedback = $aiResponse.choices[0].message.content
            Write-Host "   🤖 AI review received."
        } catch {
            $aiFeedback = "⚠️ AI review failed: $_"
        }

        # 📝 Post PR Comment
        $comment = "🤖 **AI Review for ${filePath}:**`n$aiFeedback"
        if ($syntaxIssue -or $lintOutput) {
            $comment += "`n`n⚠️ **Lint/Syntax Issues:**`n$lintOutput"
        }

        $commentBody = @{
            comments = @(@{ content = $comment; commentType = "text" })
            status = "active"
        } | ConvertTo-Json -Depth 4

        $commentUri = "$($env:AZDO_ORG_URL)/$($env:AZDO_PROJECT)/_apis/git/repositories/$($env:AZDO_REPO_ID)/pullRequests/$prId/threads?api-version=7.0"
        Invoke-RestMethod -Uri $commentUri -Headers $devOpsHeaders -Method Post -Body $commentBody

        Write-Host "   ✅ Comment added for $filePath"
    }

    Write-Host "✔ PR $prId Completed. Moving to next..."
}

Write-Host "`n🎯 All PRs processed successfully in queue order."
